cmake_minimum_required(VERSION 3.16)
project(rcauto_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(vision_msgs REQUIRED)
# Prefer distribution OpenCV so cv_bridge and our nodes use one ABI.
set(_ros_opencv_dir "/usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}/cmake/opencv4")
if(EXISTS "${_ros_opencv_dir}/OpenCVConfig.cmake")
  set(OpenCV_DIR "${_ros_opencv_dir}" CACHE PATH "Use system OpenCV with cv_bridge" FORCE)
endif()
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs videoio dnn)
find_package(cv_bridge REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2_ros REQUIRED)

add_library(rcauto_core INTERFACE)
target_include_directories(rcauto_core INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
add_executable(rcauto_controller src/controller.cpp)
ament_target_dependencies(rcauto_controller rclcpp std_msgs ackermann_msgs sensor_msgs)
target_link_libraries(rcauto_controller rcauto_core ${OpenCV_LIBS})
add_executable(rcauto_vision src/vision.cpp)
ament_target_dependencies(rcauto_vision rclcpp std_msgs sensor_msgs cv_bridge vision_msgs)
target_link_libraries(rcauto_vision rcauto_core ${OpenCV_LIBS})
add_executable(rcauto_perception src/perception.cpp)
ament_target_dependencies(rcauto_perception rclcpp std_msgs sensor_msgs vision_msgs cv_bridge)
target_link_libraries(rcauto_perception rcauto_core ${OpenCV_LIBS})
add_executable(rcauto_simulator src/simulator.cpp)
ament_target_dependencies(rcauto_simulator rclcpp std_msgs sensor_msgs vision_msgs)
target_link_libraries(rcauto_simulator ${OpenCV_LIBS})
add_executable(rcauto_gazebo_adapter src/gazebo_adapter.cpp)
ament_target_dependencies(rcauto_gazebo_adapter rclcpp ackermann_msgs geometry_msgs sensor_msgs vision_msgs nav_msgs tf2_ros)
add_executable(rcauto_f1tenth_adapter src/f1tenth_adapter.cpp)
ament_target_dependencies(rcauto_f1tenth_adapter rclcpp ackermann_msgs nav_msgs sensor_msgs std_msgs)
add_executable(rcauto_actuator src/actuator.cpp)
ament_target_dependencies(rcauto_actuator rclcpp std_msgs ackermann_msgs)
target_link_libraries(rcauto_actuator rcauto_core)
foreach(target rcauto_controller rcauto_vision rcauto_perception rcauto_simulator rcauto_actuator rcauto_gazebo_adapter rcauto_f1tenth_adapter)
  install(TARGETS ${target} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(DIRECTORY launch config sim description DESTINATION share/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(PROGRAMS
  "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/rcauto_esp32.py"
  "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/rcauto_esp32_hotplug.py"
  "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/rcauto_tuning_persist.py"
  DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(core_test test/core_test.cpp)
  target_compile_options(core_test PRIVATE -UNDEBUG)
  target_link_libraries(core_test rcauto_core)
  add_test(NAME rcauto_core_test COMMAND core_test)
  add_executable(road_vision_test test/road_vision_test.cpp)
  target_compile_options(road_vision_test PRIVATE -UNDEBUG)
  target_link_libraries(road_vision_test rcauto_core ${OpenCV_LIBS})
  add_test(NAME rcauto_road_vision_test COMMAND road_vision_test)
endif()
ament_package()
